    <script>
        // Variables de estado
        let currentTab = 'url';
        let qrcodeInstance = null;
        const qrContainer = document.getElementById("qr-container");
        const qrPlaceholder = document.getElementById("qr-placeholder");
        const btnDownload = document.getElementById("btn-download");

        function switchTab(tabId) {
            currentTab = tabId;
            
            // Ocultar todos los formularios
            document.querySelectorAll('.form-tab').forEach(tab => tab.classList.add('hidden-tab'));
            
            // Resetear estilos de botones
            document.querySelectorAll('.tab-btn').forEach(btn => {
                btn.classList.remove('bg-indigo-600', 'text-white', 'shadow-md');
                btn.classList.add('text-slate-400');
            });
            
            // Mostrar formulario activo
            document.getElementById(`tab-${tabId}`).classList.remove('hidden-tab');
            
            // Resaltar botón activo
            const activeBtn = document.getElementById(`btn-${tabId}`);
            activeBtn.classList.remove('text-slate-400');
            activeBtn.classList.add('bg-indigo-600', 'text-white', 'shadow-md');

            // Limpiar QR al cambiar para forzar al usuario a escribir
            clearQR();
            generateQR(); // Intentar generar con lo que haya en el nuevo tab
        }

        function clearQR() {
            qrContainer.innerHTML = "";
            qrContainer.style.display = "none";
            qrPlaceholder.style.display = "flex";
            btnDownload.disabled = true;
        }

        function resetColors() {
            document.getElementById('color-dark').value = "#ffffff";
            document.getElementById('color-light').value = "#000000";
            generateQR();
        }

        function generateQR() {
            let qrText = "";

            // Obtener texto según la pestaña activa
            if (currentTab === 'url') {
                qrText = document.getElementById('input-url').value.trim();
            } else if (currentTab === 'text') {
                qrText = document.getElementById('input-text').value.trim();
            } else if (currentTab === 'wifi') {
                const ssid = document.getElementById('input-wifi-ssid').value.trim();
                const pass = document.getElementById('input-wifi-pass').value.trim();
                const type = document.getElementById('input-wifi-type').value;
                if (ssid) {
                    // Formato estándar para WiFi
                    qrText = `WIFI:T:${type};S:${ssid};P:${pass};;`;
                }
            } else if (currentTab === 'whatsapp') {
                const phone = document.getElementById('input-wa-phone').value.trim().replace(/\s+/g, '');
                const msg = encodeURIComponent(document.getElementById('input-wa-msg').value.trim());
                if (phone) {
                    qrText = `https://wa.me/${phone}?text=${msg}`;
                }
            }

            // Validar si hay texto
            if (!qrText) {
                clearQR();
                return;
            }

            // Si hay texto, preparamos la UI
            qrPlaceholder.style.display = "none";
            qrContainer.style.display = "inline-block";
            btnDownload.disabled = false;
            
            // Limpiar QR anterior
            qrContainer.innerHTML = "";

            // Obtener colores
            const colorDark = document.getElementById('color-dark').value;
            const colorLight = document.getElementById('color-light').value;

            // Generar nuevo QR
            qrcodeInstance = new QRCode(qrContainer, {
                text: qrText,
                width: 256,
                height: 256,
                colorDark : colorDark,
                colorLight : colorLight,
                correctLevel : QRCode.CorrectLevel.H // Alto nivel de corrección por si hay logos o bordes
            });
        }

        function downloadQR() {
            if (btnDownload.disabled) return;

            // QRCode.js genera un canvas o un img. Buscamos ambos.
            const canvas = qrContainer.querySelector('canvas');
            const img = qrContainer.querySelector('img');
            
            let imageUrl = '';

            // Preferimos la imagen generada, si no el canvas.
            if (img && img.src && img.src.length > 100) {
                imageUrl = img.src;
            } else if (canvas) {
                imageUrl = canvas.toDataURL("image/png");
            }

            if (imageUrl) {
                const link = document.createElement('a');
                link.href = imageUrl;
                link.download = `QuickQR_${Date.now()}.png`;
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);
                
                showToast("¡QR descargado con éxito!");
            } else {
                // Pequeño truco/fix porque QRCode a veces tarda un milisegundo en dibujar el img
                setTimeout(downloadQR, 100); 
            }
        }

        function showToast(message) {
            const toast = document.getElementById('toast');
            document.getElementById('toast-msg').innerText = message;
            toast.style.opacity = '1';
            setTimeout(() => {
                toast.style.opacity = '0';
            }, 3000);
        }

        const adContainer = document.getElementById('troll-ad-container');
        const adIcon = document.getElementById('troll-ad-icon');
        const adTitle = document.getElementById('troll-ad-title');
        const adDesc = document.getElementById('troll-ad-desc');
        
        let adInterval;

        function showTrollAd() {
            // Seleccionar anuncio aleatorio
            const randomAd = trollAds[Math.floor(Math.random() * trollAds.length)];
            
            // Actualizar contenido
            adIcon.className = `fa-solid ${randomAd.icon} text-4xl mb-2 shake-animation ${randomAd.iconColor}`;
            adTitle.innerText = randomAd.title;
            adTitle.className = `font-bold text-lg leading-tight text-transparent bg-clip-text bg-gradient-to-r ${randomAd.bgGradient}`;
            adDesc.innerText = randomAd.desc;

            // Mostrar banner (quitar translate-y para que suba)
            adContainer.classList.remove('translate-y-[150%]');

            // Ocultarlo automáticamente después de 8 segundos si no lo cierra el usuario
            setTimeout(() => {
                hideTrollAd();
            }, 8000);
        }

        function hideTrollAd() {
            adContainer.classList.add('translate-y-[150%]');
        }

        // El botón de cerrar troll: a veces cierra, a veces agita el anuncio
        function closeTrollAd() {
            if (Math.random() > 0.5) {
                hideTrollAd();
                showToast("Anuncio cerrado... por ahora 😈");
            } else {
                adContainer.querySelector('.scam-ad').classList.add('shake-animation');
                setTimeout(() => {
                    adContainer.querySelector('.scam-ad').classList.remove('shake-animation');
                }, 500);
                showToast("Botón de cerrar defectuoso. ¡Intenta otra vez!");
            }
        }

        function clickTrollAd() {
            showToast("¡Casi te entra un troyano! Es broma, es un anuncio falso 🤣");
            hideTrollAd();
        }

        // Iniciar el ciclo de anuncios: Mostrar cada 20 segundos
        // Esperamos 5 segundos para mostrar el primero
        setTimeout(() => {
            showTrollAd();
            adInterval = setInterval(showTrollAd, 20000);
        }, 5000);

        // Inicializar estado ocultando el contenedor de QR
        clearQR();

    </script>
